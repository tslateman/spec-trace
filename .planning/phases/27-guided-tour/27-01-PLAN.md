4. **Add tour initialization script** (before closing `{% endblock %}`):
   ```javascript
   <script>
   function startGuidedTour() {
     const steps = [
       {
         element: '.landing-stats',
         popover: {
           title: 'Live Statistics',
           description: 'Real-time metrics update as tests run. This demo shows requirements count, pass rate, linked tests, and vendor integrations.',
           side: 'bottom',
           align: 'center'
         }
       },
       {
         element: '.feature-highlights',
         popover: {
           title: 'The SpecTrace Workflow',
           description: '1) Write specs as markdown files with YAML frontmatter. 2) Link tests with @pytest.mark.requirement("REQ-ID"). 3) View verification status in the dashboard. Click any feature card to explore.',
           side: 'top',
           align: 'center'
         }
       },
       {
         element: '.landing-paths a:first-child',
         popover: {
           title: 'Ready to Explore',
           description: 'Click Dashboard to see the traceability matrix with sample data. Green = passing tests, Red = failing, Gray = untested.',
           side: 'bottom',
           align: 'center'
         }
       }
     ];

     // Filter out stats step if no data loaded
     const statsEl = document.querySelector('.landing-stats');
     const filteredSteps = statsEl ? steps : steps.filter(s => s.element !== '.landing-stats');

     const driverObj = driver({
       showProgress: true,
       showButtons: ['next', 'previous', 'close'],
       progressText: 'Step {{current}} of {{total}}',
       nextBtnText: 'Next',
       prevBtnText: 'Previous',
       doneBtnText: 'Done',
       steps: filteredSteps
     });
     driverObj.drive();
   }

   // Auto-start tour if coming from demo hub
   document.addEventListener('DOMContentLoaded', function() {
     if (sessionStorage.getItem('startTour') === 'true') {
       sessionStorage.removeItem('startTour');
       setTimeout(startGuidedTour, 500);
     }
   });
   </script>
   ```
