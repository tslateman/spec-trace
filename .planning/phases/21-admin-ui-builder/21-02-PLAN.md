Alpine.js component (in script block):
```javascript
function flowEditor() {
  return {
    flow: {{ flow_data|safe }},
    error: '{{ error|default:"" }}',

    init() {
      // Assign unique IDs to steps for x-for tracking
      let uid = 0;
      this.flow.steps.forEach(s => s._uid = uid++);
      this.nextUid = uid;
    },

    addStep() {
      this.flow.steps.push({
        _uid: this.nextUid++,
        name: '',
        display_name: '',
        type: 'handler',
        handler: '',
        description: '',
        config: {}
      });
    },

    removeStep(index) {
      this.flow.steps.splice(index, 1);
    },

    moveUp(index) {
      if (index > 0) {
        [this.flow.steps[index], this.flow.steps[index-1]] =
        [this.flow.steps[index-1], this.flow.steps[index]];
      }
    },

    moveDown(index) {
      if (index < this.flow.steps.length - 1) {
        [this.flow.steps[index], this.flow.steps[index+1]] =
        [this.flow.steps[index+1], this.flow.steps[index]];
      }
    },

    prepareSubmit() {
      // Clean up _uid before submitting
      const cleaned = JSON.parse(JSON.stringify(this.flow));
      cleaned.steps.forEach(s => delete s._uid);
      this.$refs.flowData.value = JSON.stringify(cleaned);
    }
  };
}
```
