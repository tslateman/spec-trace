**Approach (Phase 19 scope):** Sync flows by name using the existing `update_or_create` pattern. Store metadata (source_file, requirements) in the existing `steps` JSONField as a `_metadata` key. Proper requirement M2M linking deferred to Phase 23.

```python
from dataclasses import asdict
from requirements.flows.definitions import FlowDef
from requirements.models import VerificationFlow


def sync_yaml_flows_to_db(
    flows: list[FlowDef],
    clear_existing: bool = False,
) -> dict[str, str]:
    """Sync YAML-defined flows to database.

    Args:
        flows: List of FlowDef objects from YAMLFlowParser
        clear_existing: If True, delete flows being synced before re-import

    Returns:
        Dict mapping flow name to action ('created' or 'updated')
    """
    results = {}
    flow_names = [f.name for f in flows]

    if clear_existing and flow_names:
        # Delete flows we're about to sync (allows clean re-import)
        VerificationFlow.objects.filter(name__in=flow_names).delete()

    for flow_def in flows:
        # Convert steps to list of dicts
        steps_data = [asdict(step) for step in flow_def.steps]

        # Store metadata in steps JSON (Phase 23 adds proper M2M)
        metadata = {
            '_metadata': {
                'source_file': flow_def.source_file,
                'requirements': flow_def.requirements,
                'version': flow_def.version,
            }
        }

        flow, created = VerificationFlow.objects.update_or_create(
            name=flow_def.name,
            defaults={
                'display_name': flow_def.display_name,
                'description': flow_def.description,
                'steps': [metadata] + steps_data,  # Metadata as first element
            }
        )
        results[flow_def.name] = 'created' if created else 'updated'

    return results
```

Key implementation points:
- Metadata stored as first element in steps JSON with `_metadata` key
- `clear_existing` deletes only the flows being synced (by name)
- Uses existing VerificationFlow model without schema changes
- Requirements are stored in metadata; Phase 23 adds proper linking

```python
"""Django management command for syncing YAML flow definitions to database."""
from pathlib import Path

from requirements.flows.parser import YAMLFlowParser, FlowParseError
from requirements.flows.sync import sync_yaml_flows_to_db

from .base import BaseImportCommand


class Command(BaseImportCommand):
    """Parse flow definitions from YAML files and sync to database."""

    help = 'Parse verification flow definitions from YAML files'
    path_argument_name = 'flows_dir'
    path_argument_help = 'Path to directory containing flow YAML files'

    def do_import(self, path: Path, options: dict):
        """Execute the parse and sync workflow."""
        self.stdout.write(f"Parsing flow files from {path}...")

        parser = YAMLFlowParser()

        # Collect flows, tracking errors
        flows = []
        errors = []

        for pattern in parser.FILE_PATTERNS:
            for yaml_file in sorted(path.glob(pattern)):
                try:
                    flow = parser.parse_file(yaml_file)
                    if flow:
                        flows.append(flow)
                except FlowParseError as e:
                    errors.append(str(e))
                    self.stderr.write(self.style.ERROR(f"  Error: {e}"))

        if not flows:
            self.stdout.write(self.style.WARNING("No flow files found"))
            if errors:
                self.stdout.write(self.style.ERROR(f"{len(errors)} file(s) had errors"))
            return

        self.stdout.write(f"Found {len(flows)} flow(s):")
        for flow in flows:
            req_count = len(flow.requirements)
            self.stdout.write(
                f"  - {flow.name}: {len(flow.steps)} steps, "
                f"{req_count} linked requirements"
            )

        if errors:
            self.stdout.write(self.style.WARNING(f"{len(errors)} file(s) had errors"))

        if options['dry_run']:
            self.stdout.write(self.style.SUCCESS("Dry run complete - no changes made"))
            return

        # Sync to database
        result = sync_yaml_flows_to_db(flows, clear_existing=options['clear'])

        created = sum(1 for v in result.values() if v == 'created')
        updated = sum(1 for v in result.values() if v == 'updated')

        self.stdout.write(self.style.SUCCESS(
            f"Sync complete: {created} created, {updated} updated"
        ))
```

```python
class TestYAMLFlowParser:
    """Tests for YAMLFlowParser."""

    def test_parse_valid_flow_file(self, tmp_path):
        """Parses valid YAML flow file into FlowDef."""
        yaml_content = '''
id: test-flow
title: Test Flow
description: A test flow
version: 2
requirements:
  - REQ-001
steps:
  - name: step1
    type: handler
    display_name: Step One
    handler: some.module.handler
'''
        flow_file = tmp_path / "test.yaml"
        flow_file.write_text(yaml_content)

        parser = YAMLFlowParser()
        flow = parser.parse_file(flow_file)

        assert flow is not None
        assert flow.name == "test-flow"
        assert flow.display_name == "Test Flow"
        assert flow.version == 2
        assert flow.requirements == ["REQ-001"]
        assert len(flow.steps) == 1
        assert flow.steps[0].name == "step1"
        assert flow.steps[0].type == "handler"

    def test_parse_file_without_id__returns_none(self, tmp_path):
        """Non-flow YAML files return None."""
        yaml_content = "key: value\nother: data"
        yaml_file = tmp_path / "not-flow.yaml"
        yaml_file.write_text(yaml_content)

        parser = YAMLFlowParser()
        result = parser.parse_file(yaml_file)

        assert result is None

    def test_parse_file_missing_required_field__raises_error(self, tmp_path):
        """Missing required fields raise FlowParseError."""
        yaml_content = '''
id: test-flow
# missing title and steps
'''
        flow_file = tmp_path / "bad.yaml"
        flow_file.write_text(yaml_content)

        parser = YAMLFlowParser()
        with pytest.raises(FlowParseError) as exc_info:
            parser.parse_file(flow_file)

        assert "title" in str(exc_info.value) or "steps" in str(exc_info.value)

    def test_parse_step_invalid_type__raises_error(self, tmp_path):
        """Invalid step type raises FlowParseError."""
        yaml_content = '''
id: test-flow
title: Test
steps:
  - name: bad
    type: invalid_type
    display_name: Bad Step
'''
        flow_file = tmp_path / "bad.yaml"
        flow_file.write_text(yaml_content)

        parser = YAMLFlowParser()
        with pytest.raises(FlowParseError) as exc_info:
            parser.parse_file(flow_file)

        assert "type" in str(exc_info.value).lower()

    def test_parse_handler_step_without_handler__raises_error(self, tmp_path):
        """Handler type step without handler field raises error."""
        yaml_content = '''

  - name: step1
    type: handler
    display_name: Step One
    # missing handler field
'''
        flow_file = tmp_path / "bad.yaml"

        parser = YAMLFlowParser()
        with pytest.raises(FlowParseError) as exc_info:
            parser.parse_file(flow_file)

        assert "handler" in str(exc_info.value).lower()

    def test_parse_directory__finds_all_yaml_files(self, tmp_path):
        """Parses all .yaml and .yml files in directory."""
        # Create valid flows

        assert flow.steps[0].type == "api_call"
```
